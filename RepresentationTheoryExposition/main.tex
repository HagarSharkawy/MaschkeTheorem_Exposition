% ── Master style: thick left rule + integrated title bar ────
%    Arguments: #1=frame color, #2=title-bar bg, #3=body bg
\tcbset{envstyle/.style n args={3}{%
  enhanced, breakable,
  % geometry
  arc=3pt,
  left=10pt, right=10pt,
  top=0pt, bottom=8pt,
  boxrule=0.5pt,
  % colors
  colframe=#1,
  colback=#3,
  colbacktitle=#2,
  coltitle=#1,
  % title bar sits flush inside the box, separated by a rule
 fonttitle=\bfseries\sffamily\small,
  titlerule=0.5pt,
  toptitle=4pt,
bottomtitle=4pt,
top=6pt,
  % left accent stripe
  borderline west={3pt}{0pt}{#1},
  % subtle shadow
  drop fuzzy shadow=black!12!white,
}}

% ── Shared counter ──────────────────────────────────────────
\newtcolorbox[auto counter, number within=chapter]{thmbox}[2][]{%
  envstyle={TheoremBlue}{TheoremBlueMid}{TheoremBlueLight},
  title={\textsc{Theorem}~\thetcbcounter%
         \ifx\relax#2\relax\else\enspace\textmd{(#2)}\fi},
  #1}
