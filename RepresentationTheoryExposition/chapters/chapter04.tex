\section{The Group Algebra and its Modules}

A pivotal conceptual leap in modern representation theory is the realization that linear representations of a group $G$ are in one-to-one correspondence with modules over an associative $k$-algebra, known as the \emph{group algebra} $k[G]$. This bridge allows us to apply the full machinery of ring and module theory to finite group representations.

\begin{definition}[Finite-Dimensional Associative Algebra]

A \emph{finite-dimensional algebra} $A$ over a field $k$ is a $k$-vector space equipped with a bilinear associative multiplication

\[
A \times A \longrightarrow A, \quad (a, b) \longmapsto a \cdot b
\]

possessing a multiplicative identity $1_A \in A$. Bilinearity means that for all $a, b, c \in A$ and $\lambda \in k$:

\[
(a + b)c = ac + bc, \quad a(b + c) = ab + ac, \quad (\lambda a)b = a(\lambda b) = \lambda(ab).
\]

\end{definition}



\begin{theorem}[Equivalence of Representations and Modules]

Let $G$ be a finite group and $k$ a field. There is a canonical one-to-one correspondence between:

\begin{enumerate}

    \item Linear representations of $G$ over $k$, i.e., group homomorphisms $\rho \colon G \to \operatorname{GL}(V)$;

    \item Left $k[G]$-modules $(V, \cdot)$, where the action is given by extending $\rho$ linearly:

    \[
    \left( \sum_{g \in G} a_g \, g \right) \cdot v = \sum_{g \in G} a_g \, \rho(g)v.
    \]

\end{enumerate}

Under this correspondence:

\begin{itemize}

    \item Subrepresentations of $V$ correspond precisely to $k[G]$-submodules of $V$.

    \item Irreducible representations of $G$ correspond to simple $k[G]$-modules.

    \item Intertwining operators between representations correspond to $k[G]$-module homomorphisms.

\end{itemize}

\end{theorem}

\begin{definition}[The Regular Representation]

The \emph{left regular representation} of $G$ is the vector space $V = k[G]$ equipped with the natural action of $G$ on itself by left multiplication:

\[
\rho_{\text{reg}}(g)\left(\sum_{h \in G} a_h \, h\right) = \sum_{h \in G} a_h \, (gh).
\]

Viewing $k[G]$ as a left $k[G]$-module acting on itself via ring multiplication, this is the free rank-one module $_A A$.

\end{definition}

\section{Semisimplicity and the Radical}

To understand the structure of representations, we classify modules that decompose cleanly into fundamental irreducible building blocks.

\begin{definition}[Simple and Semisimple Modules]

Let $A$ be an algebra over $k$.

\begin{enumerate}

    \item An $A$-module $V$ is \emph{simple} (or \emph{irreducible}) if $V \neq \{0\}$ and its only submodules are $\{0\}$ and $V$.

    \item An $A$-module $V$ is \emph{semisimple} (or \emph{completely reducible}) if it is isomorphic to a direct sum of simple $A$-modules:

    \[
    V \cong \bigoplus_{i \in I} V_i, \quad \text{where each } V_i \text{ is simple.}
    \]

    \item An algebra $A$ is called a \emph{semisimple algebra} if $A$, regarded as a left module over itself (denoted $_A A$), is a semisimple left $A$-module.

\end{enumerate}

\end{definition}

\begin{lemma}[Schur's Lemma]

Let $A$ be an algebra over an algebraically closed field $k$.

\begin{enumerate}

    \item If $V_1$ and $V_2$ are simple $A$-modules, then any $A$-module homomorphism $f \colon V_1 \to V_2$ is either zero or an isomorphism.

    \item If $V$ is a finite-dimensional simple $A$-module over $k$, then every $A$-module endomorphism is a scalar multiple of the identity:

    \[
    \operatorname{End}_A(V) \cong k \cdot \operatorname{id}_V \cong k.
    \]

\end{enumerate}

\end{lemma}

\begin{proof}

For (1), the kernel $\ker(f) \subseteq V_1$ is a submodule of $V_1$. Since $V_1$ is simple, either $\ker(f) = V_1$ (implying $f = 0$) or $\ker(f) = \{0\}$ (implying $f$ is injective). Similarly, the image $\operatorname{im}(f) \subseteq V_2$ is a submodule of $V_2$. Since $V_2$ is simple, either $\operatorname{im}(f) = \{0\}$ (implying $f = 0$) or $\operatorname{im}(f) = V_2$ (implying $f$ is surjective). Thus, if $f \neq 0$, $f$ is bijective, hence an isomorphism.

For (2), let $f \in \operatorname{End}_A(V)$. Since $V$ is a finite-dimensional vector space over the algebraically closed field $k$, the characteristic polynomial of $f$ has a root $\lambda \in k$. Therefore, $f - \lambda \operatorname{id}_V$ is not invertible, which means its kernel is non-trivial:

\[
\ker(f - \lambda \operatorname{id}_V) \neq \{0\}.
\]

Since $f$ and $\operatorname{id}_V$ are $A$-module homomorphisms, $f - \lambda \operatorname{id}_V \in \operatorname{End}_A(V)$. By part (1), since it is not an isomorphism, it must be the zero map. Hence $f - \lambda \operatorname{id}_V = 0$, which proves $f = \lambda \operatorname{id}_V$.

\end{proof}

\begin{definition}[The Radical]

Let $A$ be a finite-dimensional algebra over $k$. The \emph{Jacobson radical} of $A$, denoted $\operatorname{Rad}(A)$, is the intersection of the annihilators of all simple left $A$-modules:

\[
\operatorname{Rad}(A) = \left\{ a \in A \;\middle|\; a \cdot v = 0 \text{ for all } v \in V \text{ and all simple left } A\text{-modules } V \right\}.
\]

Equivalently, if $\{V_1, \dots, V_r\}$ is a complete set of representatives of isomorphism classes of simple $A$-modules with representation maps $\rho_i \colon A \to \operatorname{End}_k(V_i)$, then

\[
\operatorname{Rad}(A) = \bigcap_{i=1}^r \ker(\rho_i).
\]

\end{definition}

\begin{theorem}[Semisimplicity Criterion via Radical]

A finite-dimensional algebra $A$ is semisimple if and only if its Jacobson radical is trivial:

\[
\operatorname{Rad}(A) = \{0\}.
\]

\end{theorem}

\begin{note}

The endomorphism algebra of an algebra $A$ acting on itself as a left module is canonically anti-isomorphic to $A$:

\[
\operatorname{End}_A(_A A) \cong A^{\text{op}}.
\]

To verify this, notice that any left $A$-linear map $\phi \colon A \to A$ is uniquely determined by its value on the unit $1_A$:

\[
\phi(a) = \phi(a \cdot 1_A) = a \cdot \phi(1_A).
\]

Letting $b = \phi(1_A)$, $\phi$ is given by right multiplication by $b$: $\phi(a) = ab$. Then for two maps $\phi_b, \phi_c \in \operatorname{End}_A(A)$, their composition satisfies:

\[
(\phi_b \circ \phi_c)(1_A) = \phi_b(c) = cb,
\]

which reverses the order of multiplication, yielding the opposite algebra $A^{\text{op}}$.

\end{note}

\section{The Density Theorem and Artin-Wedderburn Theorem}

\label{sec:density_theorem}

We now address one of the deepest results from the notes: the Jacobson Density Theorem, which shows that the representation map $\bigoplus_{i=1}^r \rho_i \colon A \to \bigoplus_{i=1}^r \operatorname{End}_k(V_i)$ is surjective.

\begin{lemma}[ ]

\label{lem:vector_realization}

Let $A$ be an algebra over an algebraically closed field $k$, and let $V$ be a finite-dimensional simple $A$-module. If $\{v_1, v_2, \dots, v_n\} \subset V$ is any linearly independent set of vectors, then for any arbitrary vectors $w_1, w_2, \dots, w_n \in V$, there exists an element $a \in A$ such that

\[
a \cdot v_i = w_i \quad \text{for all } 1 \le i \le n.
\]

\end{lemma}

\begin{proof}

Consider the direct sum $V^{\oplus n} = \underbrace{V \oplus \dots \oplus V}_{n \text{ times}}$. We define an $A$-module homomorphism

\[
\Psi \colon A \longrightarrow V^{\oplus n}, \quad a \longmapsto (a v_1, a v_2, \dots, a v_n).
\]

Its image $U = \operatorname{im}(\Psi) \subseteq V^{\oplus n}$ is an $A$-submodule of $V^{\oplus n}$. We must show that $U = V^{\oplus n}$, for if $U = V^{\oplus n}$, then $(w_1, \dots, w_n) \in U$, meaning there exists $a \in A$ with $\Psi(a) = (w_1, \dots, w_n)$, i.e., $a v_i = w_i$ for all $i$.

Since $V$ is a simple $A$-module, the direct sum $V^{\oplus n}$ is a semisimple $A$-module. By general module theory, every submodule of a semisimple module is a direct summand and isomorphic to a direct sum of a subset of the simple summands. Hence, there exists an integer $r \le n$ such that

\[
U \cong V^{\oplus r}.
\]

We claim that $r = n$. Suppose for contradiction that $r < n$. 

The inclusion map $\phi \colon U \hookrightarrow V^{\oplus n}$ is an $A$-module homomorphism from $V^{\oplus r}$ to $V^{\oplus n}$. By Schur's Lemma, since $\operatorname{End}_A(V) \cong k$, any $A$-module homomorphism between $V^{\oplus r}$ and $V^{\oplus n}$ is represented by an $r \times n$ matrix $X \in \operatorname{Mat}_{r \times n}(k)$:

\[
\phi(u_1, \dots, u_r) = (u_1, \dots, u_r) X.
\]

Since $r < n$, the rank of $X$ is at most $r < n$. In particular, the column rank of $X$ is strictly less than $n$, which implies that the right null space of $X$ is non-trivial. That is, there exists a non-zero vector $q = (q_1, \dots, q_n)^T \in k^n$ such that

\[
X \begin{pmatrix} q_1 \\\ \vdots \\\ q_n \end{pmatrix} = \begin{pmatrix} 0 \\\ \vdots \\\ 0 \end{pmatrix}.
\]

Now, observe that the element $1_A \in A$ maps under $\Psi$ to $(v_1, \dots, v_n) \in U$. Because $(v_1, \dots, v_n) \in U$, there exists some $(u_1, \dots, u_r) \in V^{\oplus r}$ such that

\[
(v_1, \dots, v_n) = (u_1, \dots, u_r) X.
\]

Multiplying both sides on the right by the column vector $q$:

\[
\sum_{i=1}^n q_i v_i = (v_1, \dots, v_n) \begin{pmatrix} q_1 \\\ \vdots \\\ q_n \end{pmatrix} = (u_1, \dots, u_r) X \begin{pmatrix} q_1 \\\ \vdots \\\ q_n \end{pmatrix} = (u_1, \dots, u_r) \cdot 0 = 0.
\]

This produces a non-trivial linear combination $\sum_{i=1}^n q_i v_i = 0$ with not all $q_i = 0$. But this directly contradicts the hypothesis that $\{v_1, \dots, v_n\}$ is linearly independent!

Therefore, the assumption $r < n$ is false. We must have $r = n$, which forces $U = V^{\oplus n}$. Hence, the map $\Psi$ is surjective, and there exists $a \in A$ such that $a v_i = w_i$ for each $i \in \{1, \dots, n\}$.

\end{proof}

\begin{theorem}[The Density Theorem for Finite-Dimensional Algebras]

\label{thm:density}

Let $A$ be an algebra over an algebraically closed field $k$.

\begin{enumerate}

    \item If $V$ is a finite-dimensional simple $A$-module, the representation map

    \[
    \rho \colon A \longrightarrow \operatorname{End}_k(V)
    \]

    is surjective.

    \item If $V_1, V_2, \dots, V_r$ are pairwise non-isomorphic finite-dimensional simple $A$-modules, the direct sum representation map

    \[
    \rho_1 \oplus \dots \oplus \rho_r \colon A \longrightarrow \bigoplus_{i=1}^r \operatorname{End}_k(V_i)
    \]

    is surjective.

\end{enumerate}

\end{theorem}

\begin{proof}

\textbf{Part (1):} Let $d = \dim_k(V)$, and let $\{v_1, \dots, v_d\}$ be a vector space basis of $V$. Let $C \in \operatorname{End}_k(V)$ be any arbitrary linear operator on $V$. We want to find an element $a \in A$ such that $\rho(a) = C$.

Define the vectors $w_i = C(v_i) \in V$ for each $1 \le i \le d$. Since $\{v_1, \dots, v_d\}$ is linearly independent in $V$, Lemma~\ref{lem:vector_realization} guarantees the existence of an element $a \in A$ such that

\[
a \cdot v_i = w_i = C(v_i) \quad \text{for all } 1 \le i \le d.
\]

Since a linear transformation is entirely determined by its action on a basis, $\rho(a)v = C(v)$ for all $v \in V$, meaning $\rho(a) = C$. Thus $\rho$ is surjective.

\medskip

\textbf{Part (2):} Let $B \subseteq \bigoplus_{i=1}^r \operatorname{End}_k(V_i)$ be the image of $A$ under $\bigoplus_{i=1}^r \rho_i$. Notice that each $\operatorname{End}_k(V_i)$, regarded as an $A$-module via the action $a \cdot M = \rho_i(a) M$, is isomorphic to $V_i^{\oplus d_i}$ where $d_i = \dim_k(V_i)$ (since matrix multiplication acts independently on each of the $d_i$ columns).

Because $V_1, \dots, V_r$ are simple and pairwise non-isomorphic, the ambient space $\bigoplus_{i=1}^r \operatorname{End}_k(V_i)$ is a semisimple $A$-module with isotypic components $\operatorname{End}_k(V_i) \cong V_i^{\oplus d_i}$.

The image $B \subseteq \bigoplus_{i=1}^r \operatorname{End}_k(V_i)$ is an $A$-submodule. By semisimplicity, $B$ decomposes as

\[
B = \bigoplus_{i=1}^r (B \cap \operatorname{End}_k(V_i)).
\]

Let $\pi_i \colon \bigoplus_{j=1}^r \operatorname{End}_k(V_j) \to \operatorname{End}_k(V_i)$ be the canonical projection onto the $i$-th component. By Part (1), the projection $\pi_i(B) = \rho_i(A) = \operatorname{End}_k(V_i)$ is surjective for every $i$. Since there are no non-zero $A$-homomorphisms between $V_i$ and $V_j$ for $i \neq j$ (by Schur's Lemma), any submodule $B$ whose projections cover each factor must be the entire direct sum:

\[
B = \bigoplus_{i=1}^r \operatorname{End}_k(V_i).
\]

Therefore, $\bigoplus_{i=1}^r \rho_i$ is surjective.

\end{proof}

\begin{corollary}[Artin-Wedderburn Structure Theorem]

\label{cor:artin_wedderburn}

Let $A$ be a finite-dimensional algebra over an algebraically closed field $k$. Then:

\begin{enumerate}

    \item $A$ has only finitely many pairwise non-isomorphic simple modules $V_1, \dots, V_r$.

    \item There is a canonical isomorphism of algebras:

    \[
    A / \operatorname{Rad}(A) \cong \bigoplus_{i=1}^r \operatorname{End}_k(V_i) \cong \bigoplus_{i=1}^r M_{d_i}(k), \quad \text{where } d_i = \dim_k(V_i).
    \]

    \item In particular, if $A$ is semisimple ($\operatorname{Rad}(A) = \{0\}$), then $A$ is isomorphic to a direct sum of matrix algebras:

    \[
    A \cong \bigoplus_{i=1}^r M_{d_i}(k).
    \]

\end{enumerate}

\end{corollary}

\begin{proof}

Consider any collection of pairwise non-isomorphic simple $A$-modules $V_1, \dots, V_r$. By Theorem~\ref{thm:density}, the map

\[
\bigoplus_{i=1}^r \rho_i \colon A \longrightarrow \bigoplus_{i=1}^r \operatorname{End}_k(V_i)
\]

is surjective. Its kernel is $\bigcap_{i=1}^r \ker(\rho_i) = \operatorname{Rad}(A)$. By the First Isomorphism Theorem for algebras:

\[
A / \operatorname{Rad}(A) \cong \bigoplus_{i=1}^r \operatorname{End}_k(V_i) \cong \bigoplus_{i=1}^r M_{d_i}(k).
\]

Taking dimensions over $k$:

\[
\dim_k(A) \ge \dim_k(A / \operatorname{Rad}(A)) = \sum_{i=1}^r d_i^2.
\]

Since each $d_i \ge 1$ and $\dim_k(A) < \infty$, the number of pairwise non-isomorphic simple modules $r$ is strictly bounded by $\dim_k(A)$:

\[
r \le \sum_{i=1}^r d_i^2 \le \dim_k(A) < \infty.
\]

Thus, there can only be finitely many simple modules. If $A$ is semisimple, $\operatorname{Rad}(A) = \{0\}$, immediately yielding $A \cong \bigoplus_{i=1}^r M_{d_i}(k)$.

\end{proof}

\section{Maschke's Theorem}

\label{sec:maschke}

Having established the structure of semisimple algebras, we now address the cornerstone theorem that connects group theory to algebra semisimplicity: Maschke's Theorem.

\begin{theorem}[Maschke's Theorem]

Let $G$ be a finite group, and let $k$ be a field whose characteristic does not divide the order of $G$ (i.e., $\operatorname{char}(k) \nmid |G|$). Then every finite-dimensional representation of $G$ over $k$ is completely reducible (semisimple). 

In particular, the group algebra $k[G]$ is a semisimple algebra.

\end{theorem}

\begin{proof}

Let $V$ be a finite-dimensional representation of $G$ over $k$, and let $W \subseteq V$ be a $G$-invariant subspace (subrepresentation). To prove that $V$ is semisimple, it suffices to show that $W$ admits a $G$-invariant complement $W'$, so that $V = W \oplus W'$.

\medskip

\textbf{Step 1: Choose a vector space projection.}

By elementary linear algebra, choose any vector space complement $U$ of $W$ in $V$, so $V = W \oplus U$. Let $P \colon V \to W$ be the canonical vector space projection of $V$ onto $W$ along $U$. By definition:

\begin{itemize}

    \item $P(v) \in W$ for all $v \in V$;

    \item $P(w) = w$ for all $w \in W$ (i.e., $P|_W = \operatorname{id}_W$);

    \item $P^2 = P$.

\end{itemize}

However, this projection $P$ is generally \emph{not} a $G$-homomorphism; that is, $P(\rho(g)v) \neq \rho(g)P(v)$ in general.

\medskip

\textbf{Step 2: Construct the Reynolds Averaging Operator.}

To construct a $G$-equivariant projection, we average the conjugates of $P$ over the entire group $G$. Define the operator $\bar{P} \colon V \to V$ by:

\[
\bar{P} = \frac{1}{|G|} \sum_{g \in G} \rho(g) \circ P \circ \rho(g^{-1}).
\]

Here, the hypothesis $\operatorname{char}(k) \nmid |G|$ is critical: it guarantees that $|G| \cdot 1_k \neq 0$ in the field $k$, making the scalar $\frac{1}{|G|} \in k$ well-defined and invertible.

\medskip

\textbf{Step 3: Verify that $\bar{P}$ is a $G$-homomorphism.}

Let $h \in G$. We compute the conjugation of $\bar{P}$ by $\rho(h)$:

\begin{align*}
\rho(h) \circ \bar{P} \circ \rho(h^{-1}) &= \rho(h) \left( \frac{1}{|G|} \sum_{g \in G} \rho(g) \circ P \circ \rho(g^{-1}) \right) \rho(h^{-1}) \\
&= \frac{1}{|G|} \sum_{g \in G} \rho(h) \rho(g) \circ P \circ \rho(g^{-1}) \rho(h^{-1}) \\
&= \frac{1}{|G|} \sum_{g \in G} \rho(hg) \circ P \circ \rho((hg)^{-1}).
\end{align*}

As $g$ ranges over all elements of $G$, the product $x = hg$ also ranges over all elements of $G$ exactly once (since left translation by $h$ is a bijection of $G$). Therefore:

\[
\rho(h) \circ \bar{P} \circ \rho(h^{-1}) = \frac{1}{|G|} \sum_{x \in G} \rho(x) \circ P \circ \rho(x^{-1}) = \bar{P}.
\]

Multiplying both sides on the right by $\rho(h)$ yields:

\[
\rho(h) \circ \bar{P} = \bar{P} \circ \rho(h) \quad \text{for all } h \in G.
\]

Thus, $\bar{P} \in \operatorname{End}_G(V)$ is an intertwining operator.

\medskip

\textbf{Step 4: Verify that $\bar{P}$ is a projection onto $W$.}

First, take any vector $v \in V$. For each $g \in G$, $P(\rho(g^{-1})v) \in W$ because the image of $P$ is $W$. Because $W$ is a $G$-invariant subspace, applying $\rho(g)$ preserves $W$:

\[
\rho(g) \left( P(\rho(g^{-1})v) \right) \in W.
\]

Thus, every term in the sum defining $\bar{P}(v)$ lies in $W$. Because $W$ is a subspace closed under linear combinations, $\bar{P}(v) \in W$ for all $v \in V$. Hence $\operatorname{im}(\bar{P}) \subseteq W$.

Next, take any vector $w \in W$. Since $W$ is $G$-invariant, $\rho(g^{-1})w \in W$. Because $P|_W = \operatorname{id}_W$, we have $P(\rho(g^{-1})w) = \rho(g^{-1})w$. Applying $\rho(g)$:

\[
\rho(g) \left( P(\rho(g^{-1})w) \right) = \rho(g) \rho(g^{-1}) w = \rho(e) w = w.
\]

Substituting this into the definition of $\bar{P}$:

\[
\bar{P}(w) = \frac{1}{|G|} \sum_{g \in G} w = \frac{1}{|G|} \cdot |G| w = w.
\]

Therefore, $\bar{P}|_W = \operatorname{id}_W$. Since $\operatorname{im}(\bar{P}) \subseteq W$ and $\bar{P}(w) = w$ for all $w \in W$, it follows that $\bar{P}^2 = \bar{P}$ and $\operatorname{im}(\bar{P}) = W$. Thus $\bar{P}$ is a projection onto $W$.

\medskip

\textbf{Step 5: Define the complementary subrepresentation.}

Set $W' = \ker(\bar{P})$. Since $\bar{P}$ is a projection onto $W$, basic linear algebra implies

\[
V = \operatorname{im}(\bar{P}) \oplus \ker(\bar{P}) = W \oplus W'.
\]

Finally, we show that $W'$ is $G$-invariant. Let $w' \in W'$ and $g \in G$. Then $\bar{P}(w') = 0$. Using the intertwining property from Step 3:

\[
\bar{P}(\rho(g)w') = \rho(g)\bar{P}(w') = \rho(g)(0) = 0.
\]

Hence $\rho(g)w' \in \ker(\bar{P}) = W'$. Therefore, $W'$ is a $G$-invariant subspace.

By induction on dimension, every finite-dimensional representation decomposes into a direct sum of irreducible representations. In particular, applying this to the left regular representation $V = k[G]$, the algebra $k[G]$ decomposes as a direct sum of simple left $k[G]$-modules, which by definition means $k[G]$ is a semisimple algebra.

\end{proof}

\begin{corollary}[Decomposition of the Group Algebra]

Let $G$ be a finite group and $k$ an algebraically closed field such that $\operatorname{char}(k) \nmid |G|$. Let $V_1, \dots, V_r$ be the complete set of pairwise non-isomorphic irreducible representations of $G$, and let $d_i = \dim_k(V_i)$. Then:

\begin{enumerate}

    \item There is an algebra isomorphism:

    \[
    k[G] \cong \bigoplus_{i=1}^r \operatorname{End}_k(V_i) \cong \bigoplus_{i=1}^r M_{d_i}(k).
    \]

    \item The order of the group equals the sum of the squares of the degrees of the irreducible representations:

    \[
    |G| = \sum_{i=1}^r d_i^2 = d_1^2 + d_2^2 + \dots + d_r^2.
    \]

    \item The regular representation decomposes as:

    \[
    k[G] \cong \bigoplus_{i=1}^r d_i V_i = \bigoplus_{i=1}^r V_i^{\oplus d_i}.
    \]

    That is, each irreducible representation $V_i$ appears in the regular representation with multiplicity equal to its dimension $d_i$.

\end{enumerate}

\end{corollary}

\section{The Modular Case: When Maschke's Theorem Fails}

\label{sec:modular_failure}

In the lecture notes, considerable attention was dedicated to the question: \emph{"Why does this fail when $\operatorname{char}(k) \mid |G|$?"} The study of representations when the field characteristic divides the group order is called \emph{modular representation theory}. In this regime, group algebras are generally \textbf{not} semisimple, and representations fail to be completely reducible.

\begin{remark}
\textbf{The Canonical Counterexample: $G = \mathbb{Z}_2$ over $\mathbb{F}_2$.}


Let $G = \mathbb{Z}_2 = \{e, g\}$ with $g^2 = e$, and let $k = \mathbb{F}_2 = \{0, 1\}$ be the finite field with two elements. Here $|G| = 2$, so $\operatorname{char}(k) = 2$ divides $|G|$.

The regular representation $V = k[\mathbb{Z}_2]$ is a 2-dimensional vector space with basis $\{e, g\}$. The generator $g$ acts on this basis by:

\[
g \cdot e = g, \quad g \cdot g = e.
\]

In the ordered basis $(e, g)$, the matrix representing $g$ is

\[
\rho(g) = \begin{pmatrix} 0 & 1 \\\ 1 & 0 \end{pmatrix}.
\]

Notice that over $\mathbb{F}_2$:

\[
(\rho(g) - I)^2 = \rho(g)^2 - 2\rho(g) + I = I - 0 + I = 2I = 0.
\]

Consider the 1-dimensional subspace:

\[
W = \operatorname{span}_{\mathbb{F}_2}\{e + g\}.
\]

Since $g \cdot (e + g) = g + e = e + g$, $W$ is a 1-dimensional $G$-invariant subrepresentation (isomorphic to the trivial representation).

Now, does $W$ have a $G$-invariant complement $W'$ in $V$?

If $W'$ were an invariant complement, $\dim_{\mathbb{F}_2}(W') = 1$, so $W' = \operatorname{span}_{\mathbb{F}_2}\{v\}$ for some non-zero vector $v \notin W$. The remaining non-zero vectors in $\mathbb{F}_2^2$ are:

\[
e \quad \text{and} \quad g.
\]

However:

\[
g \cdot e = g \notin \operatorname{span}\{e\}, \quad g \cdot g = e \notin \operatorname{span}\{g\}.
\]

Neither $\operatorname{span}\{e\}$ nor $\operatorname{span}\{g\}$ is invariant under the action of $g$!

Indeed, any 1-dimensional invariant subspace must be spanned by an eigenvector of $\rho(g)$. The characteristic polynomial of $\rho(g)$ is

\[
\det(tI - \rho(g)) = t^2 - 1 = (t - 1)^2 \in \mathbb{F}_2[t].
\]

The eigenspace for $\lambda = 1$ is the null space of $\rho(g) - I = \begin{pmatrix} 1 & 1 \\\ 1 & 1 \end{pmatrix}$, which is exactly $W = \operatorname{span}\{e + g\}$, of dimension $1$.

Changing to the basis $v_1 = e + g$ and $v_2 = e$, the matrix of $\rho(g)$ becomes:

\[
P^{-1} \rho(g) P = \begin{pmatrix} 1 & 1 \\\ 0 & 1 \end{pmatrix},
\]

which is a \textbf{single Jordan block of size 2}!

Consequently:

\begin{itemize}

    \item $V$ is \emph{indecomposable}: it cannot be split into a direct sum $W \oplus W'$.

    \item $V$ is \emph{not irreducible}: it contains the proper subrepresentation $W$.

    \item Therefore, $k[\mathbb{Z}_2]$ is \textbf{not semisimple}.

\end{itemize}

The averaging operator fails because:

\[
\sum_{h \in G} \rho(h) P \rho(h^{-1}) = P + \rho(g) P \rho(g),
\]

which has trace zero in characteristic 2, and dividing by $|G| = 2 \equiv 0 \pmod 2$ is undefined.

\end{remark}

\section{Concrete Examples and Applications}

\subsection{The Cyclic Group $\mathbb{Z}_2$ in Characteristic $\neq 2$}

When $\operatorname{char}(k) \neq 2$, the group algebra $k[\mathbb{Z}_2]$ decomposes according to Maschke's Theorem. Since $\mathbb{Z}_2$ is abelian, all irreducible representations are 1-dimensional. The condition $|G| = \sum d_i^2$ becomes $2 = 1^2 + 1^2$, giving exactly two non-isomorphic 1-dimensional representations:

\begin{enumerate}

    \item \textbf{Trivial representation}: $\rho_{\text{triv}}(g) = [1]$.

    \item \textbf{Sign representation}: $\rho_{\text{sgn}}(g) = [-1]$.

\end{enumerate}

The idempotents that project onto these subrepresentations are:

\[
e_{\text{triv}} = \frac{e + g}{2}, \quad e_{\text{sgn}} = \frac{e - g}{2}.
\]

We readily check that $e_{\text{triv}}^2 = e_{\text{triv}}$, $e_{\text{sgn}}^2 = e_{\text{sgn}}$, and $e_{\text{triv}} e_{\text{sgn}} = 0$, giving the algebra isomorphism:

\[
k[\mathbb{Z}_2] \cong k \oplus k.
\]

\subsection{The Symmetric Group $S_3$}

The symmetric group on 3 elements, $S_3 = \langle \sigma, \tau \mid \sigma^3 = 1, \, \tau^2 = 1, \, \tau \sigma \tau = \sigma^{-1} \rangle$, has order $|S_3| = 6$.

Over $\mathbb{C}$, the dimension formula requires:

\[
6 = \sum_{i=1}^r d_i^2.
\]

The number of conjugacy classes of $S_3$ is 3 (corresponding to cycle types $(\cdot)(\cdot)(\cdot)$, $(\cdot\,\cdot)(\cdot)$, and $(\cdot\,\cdot\,\cdot)$), so $r = 3$. The unique positive integer solution to $6 = d_1^2 + d_2^2 + d_3^2$ is

\[
6 = 1^2 + 1^2 + 2^2.
\]

The three irreducible representations of $S_3$ are:

\begin{enumerate}

    \item \textbf{Trivial Representation} $V_1$ ($d_1 = 1$): $\rho_{\text{triv}}(\pi) = 1$ for all $\pi \in S_3$.

    \item \textbf{Sign Representation} $V_2$ ($d_2 = 1$): $\rho_{\text{sgn}}(\pi) = \operatorname{sgn}(\pi) \in \{\pm 1\}$.

    \item \textbf{Standard 2D Representation} $V_3$ ($d_3 = 2$): 

    Consider the natural permutation representation on $k^3$ with basis $\{e_1, e_2, e_3\}$. The 1-dimensional subspace spanned by $e_1 + e_2 + e_3$ is invariant and carries the trivial representation. Its orthogonal complement

    \[
    V_3 = \left\{ (x_1, x_2, x_3) \in k^3 \;\middle|\; x_1 + x_2 + x_3 = 0 \right\}
    \]

    is a 2-dimensional irreducible representation. Choosing a basis $u_1 = e_1 - e_2$ and $u_2 = e_2 - e_3$, the generators $\sigma = (1\,2\,3)$ and $\tau = (1\,2)$ act by:

    \[
    \rho_3(\sigma) = \begin{pmatrix} 0 & -1 \\\ 1 & -1 \end{pmatrix}, \quad \rho_3(\tau) = \begin{pmatrix} -1 & 1 \\\ 0 & 1 \end{pmatrix}.
    \]

\end{enumerate}

By the Artin-Wedderburn Theorem:

\[
\mathbb{C}[S_3] \cong \mathbb{C} \oplus \mathbb{C} \oplus M_2(\mathbb{C}).
\]

The regular representation decomposes as:

\[
\mathbb{C}[S_3] \cong V_{\text{triv}} \oplus V_{\text{sgn}} \oplus V_3^{\oplus 2},
\]

illustrating that the 2-dimensional representation occurs with multiplicity 2 in the regular representation.

\section{Exercises}

\begin{exercise}[The Center of the Group Algebra]

Let $G$ be a finite group and $k$ an algebraically closed field with $\operatorname{char}(k) \nmid |G|$. Show that the dimension of the center of the group algebra $Z(k[G])$ equals the number of conjugacy classes of $G$, and deduce that the number of pairwise non-isomorphic irreducible representations of $G$ equals the number of conjugacy classes of $G$.

\end{exercise}

\begin{exercise}[Dual Representation]

Let $\rho \colon G \to \operatorname{GL}(V)$ be a representation of $G$. Define $\rho^\ast \colon G \to \operatorname{GL}(V^\ast)$ on the dual space $V^\ast = \operatorname{Hom}_k(V, k)$ by

\[
(\rho^\ast(g)f)(v) = f(\rho(g^{-1})v) \quad \text{for } f \in V^\ast, v \in V.
\]

Prove that $\rho^\ast$ is a representation of $G$, and show that $V^\ast$ is irreducible if and only if $V$ is irreducible.

\end{exercise}
